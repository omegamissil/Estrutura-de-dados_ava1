#include <stdio.h>

#define TOTAL_ALUNOS 6
#define TOTAL_COLUNAS 3

void processarAlunos(float dados[TOTAL_ALUNOS][TOTAL_COLUNAS], float medias[TOTAL_ALUNOS]) {
    
    for (int i = 0; i < TOTAL_ALUNOS; i++) {
        printf("\nAluno %d:\n", i + 1);
        
        printf("Digite a primeira nota: ");
        scanf("%f", &dados[i][0]);
        
        printf("Digite a segunda nota: ");
        scanf("%f", &dados[i][1]);
        
        dados[i][2] = (dados[i][0] + dados[i][1]) / 2.0;
        medias[i] = dados[i][2];
        
        printf("Média do Aluno %d: %.2f\n", i + 1, dados[i][2]);
    }
}

void analisarTurma(float medias[TOTAL_ALUNOS], float *mediaGeral, int *acima, int *abaixo) {
    float somaGeral = 0.0;
    
    for (int i = 0; i < TOTAL_ALUNOS; i++) {
        somaGeral += medias[i];
    }
    
    *mediaGeral = somaGeral / TOTAL_ALUNOS;

    *acima = 0;
    *abaixo = 0;

    for (int i = 0; i < TOTAL_ALUNOS; i++) {
        if (medias[i] >= *mediaGeral) {
            (*acima)++;
        } else {
            (*abaixo)++;
        }
    }
}

int main() {
    float dados[TOTAL_ALUNOS][TOTAL_COLUNAS];
    float medias[TOTAL_ALUNOS];
    
    float mediaGeral = 0.0;
    int alunosAcima = 0, alunosAbaixo = 0;
    processarAlunos(dados, medias);
    analisarTurma(medias, &mediaGeral, &alunosAcima, &alunosAbaixo);
    printf("\n\n");
    printf("Relatório geral de notas e médias dos alunos: \n");
    printf("\n");
    
    for (int i = 0; i < TOTAL_ALUNOS; i++) {
        
        printf("Aluno %d -> Nota 1: %.2f | Nota 2: %.2f | Média: %.2f\n", 
               i + 1, dados[i][0], dados[i][1], dados[i][2]);
    }
    
    printf("\n\n");
    printf("Média geral da turma: %.2f\n", mediaGeral);
    printf("Quantidade de alunos acima da media: %d\n", alunosAcima);
    printf("Quantidade de alunos abaixo da media: %d\n", alunosAbaixo);

    return 0;
}
